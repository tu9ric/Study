cmake_minimum_required(VERSION 3.16)
project(lab1 LANGUAGES CXX)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("int main() { __int128 n = 0; return static_cast<int>(n); }" HAS_INT128)
if(NOT HAS_INT128)
    message(FATAL_ERROR "Lab 1 requires 64-bit GCC/MinGW-w64 with __int128 support. On Windows use MSYS2 UCRT64, not MSVC.")
endif()
find_package(Threads REQUIRED)
add_executable(lab1_task1 task1.cpp)
add_executable(lab1_task2 task2.cpp)
foreach(target IN ITEMS lab1_task1 lab1_task2)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_link_libraries(${target} PRIVATE Threads::Threads)
endforeach()
target_compile_options(lab1_task1 PRIVATE -mavx2)

# Keep source literals and output bytes in UTF-8 on both platforms.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(lab1_task1 PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(lab1_task2 PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
